Models Module

The chemxai.models module provides neural network architectures for molecular property prediction on both tabular and graph-based data.

Overview

This module contains two main model classes:

  • MLP: Multi-Layer Perceptron for tabular molecular descriptors
  • GCN: Graph Convolutional Network for molecular graphs

MLP - Multi-Layer Perceptron

Fully connected neural network for regression tasks on tabular molecular data.

Class Definition

class MLP(nn.Module):
    """
    Multi-Layer Perceptron for molecular property prediction.
    
    Parameters
    ----------
    input_dim : int
        Dimension of input features (e.g., number of fingerprint bits)
    output_dim : int
        Dimension of output (typically 1 for regression)
    layers : list of int
        List specifying the number of neurons in each hidden layer
        Example: [64, 32] creates two hidden layers with 64 and 32 neurons
    device : torch.device
        Device to run the model on ('cpu' or 'cuda')
    lr : float, optional (default=0.001)
        Learning rate for Adam optimizer
        
    Attributes
    ----------
    layers : nn.Sequential
        Sequential container of all layers
    criterion : nn.L1Loss
        Mean Absolute Error loss function
    optimizer : torch.optim.Adam
        Adam optimizer with specified learning rate
    device : torch.device
        Device where model resides
    
    Example
    -------
    >>> import torch
    >>> from chemxai.models import MLP
    >>> 
    >>> # Create MLP for Morgan fingerprints (2048 bits)
    >>> device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')
    >>> model = MLP(
    ...     input_dim=2048,
    ...     output_dim=1,
    ...     layers=[256, 128, 64],
    ...     device=device,
    ...     lr=0.001
    ... )
    >>> 
    >>> # Forward pass
    >>> x = torch.randn(32, 2048).to(device)  # batch of 32 molecules
    >>> predictions = model(x)
    >>> print(predictions.shape)  # torch.Size([32, 1])
    """
    
    def __init__(self, input_dim, output_dim, layers, device, lr=0.001):
        super(MLP, self).__init__()
        all_layers = []
        prev_dim = input_dim

        # Hidden layers with ReLU activation
        for layer_dim in layers:
            all_layers.append(nn.Linear(prev_dim, layer_dim))
            all_layers.append(nn.ReLU())
            prev_dim = layer_dim

        # Output layer (no activation for regression)
        all_layers.append(nn.Linear(prev_dim, output_dim))

        self.layers = nn.Sequential(*all_layers)
        self.criterion = nn.L1Loss()  # Mean Absolute Error
        self.optimizer = optim.Adam(self.parameters(), lr=lr)
        self.device = device

    def forward(self, x):
        """
        Forward pass through the network.
        
        Parameters
        ----------
        x : torch.Tensor
            Input features of shape [batch_size, input_dim]
            
        Returns
        -------
        torch.Tensor
            Predictions of shape [batch_size, output_dim]
        """
        return self.layers(x)

Usage Examples

Basic Training Loop

import torch
import torch.nn.functional as F
from chemxai.models import MLP
from chemxai.data import qm9_tabular

# Setup
device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')

# Load data
qm9 = qm9_tabular()
train_loader, val_loader, test_loader, *_ = qm9.get_paired_dataloaders_tabular(
    att_index=7,  # Energy gap
    batch_size=64,
    descriptor_type='Morgan',
    morgan_nBits=2048
)

# Create model
model = MLP(
    input_dim=2048,
    output_dim=1,
    layers=[256, 128, 64],
    device=device,
    lr=0.001
).to(device)

# Training loop
model.train()
for epoch in range(100):
    total_loss = 0
    for batch_x, batch_y in train_loader:
        batch_x, batch_y = batch_x.to(device), batch_y.to(device)
        
        # Forward pass
        model.optimizer.zero_grad()
        predictions = model(batch_x)
        loss = model.criterion(predictions, batch_y)
        
        # Backward pass
        loss.backward()
        model.optimizer.step()
        
        total_loss += loss.item()
    
    # Validation
    model.eval()
    val_loss = 0
    with torch.no_grad():
        for batch_x, batch_y in val_loader:
            batch_x, batch_y = batch_x.to(device), batch_y.to(device)
            predictions = model(batch_x)
            val_loss += model.criterion(predictions, batch_y).item()
    
    print(f'Epoch {epoch}: Train Loss = {total_loss/len(train_loader):.4f}, '
          f'Val Loss = {val_loss/len(val_loader):.4f}')
    model.train()

# Save model
torch.save(model.state_dict(), 'mlp_model.pth')

GCN - Graph Convolutional Network

Graph neural network for molecular property prediction from molecular graphs.

Class Definition

class GCN(torch.nn.Module):
    """
    Graph Convolutional Network for molecular graphs.
    
    Architecture:
    - 3 GCN layers with batch normalization and ReLU activation
    - Global add pooling for graph-level representation
    - 2 fully connected layers for final prediction
    - Dropout (0.3) for regularization
    
    Parameters
    ----------
    num_features : int
        Number of input node features
    hidden_dim : int, optional (default=256)
        Dimension of hidden layers in GCN and FC layers
        
    Example
    -------
    >>> import torch
    >>> from chemxai.models import GCN
    >>> from chemxai.data import graph_datasets
    >>> 
    >>> # Load graph data
    >>> gd = graph_datasets()
    >>> dataset = gd.prepare_data_graph('QM9')
    >>> 
    >>> # Create model
    >>> model = GCN(num_features=dataset[0].x.shape[1], hidden_dim=256)
    >>> 
    >>> # Forward pass for single graph
    >>> data = dataset[0]
    >>> prediction = model(data.x, data.edge_index)
    >>> print(f'Prediction shape: {prediction.shape}')
    >>> 
    >>> # Forward pass for batch
    >>> from torch_geometric.loader import DataLoader
    >>> loader = DataLoader(dataset, batch_size=32, shuffle=True)
    >>> batch = next(iter(loader))
    >>> predictions = model(batch.x, batch.edge_index, batch.batch)
    >>> print(f'Batch predictions shape: {predictions.shape}')
    """
    
    def __init__(self, num_features, hidden_dim=256):
        super(GCN, self).__init__()
        
        # Graph convolutional layers
        self.conv1 = GCNConv(num_features, hidden_dim)
        self.bn1 = BatchNorm1d(hidden_dim)
        
        self.conv2 = GCNConv(hidden_dim, hidden_dim)
        self.bn2 = BatchNorm1d(hidden_dim)
        
        self.conv3 = GCNConv(hidden_dim, hidden_dim)
        self.bn3 = BatchNorm1d(hidden_dim)

        # Fully connected layers
        self.lin1 = Linear(hidden_dim, hidden_dim)
        self.lin2 = Linear(hidden_dim, 1)
        
        # Dropout for regularization
        self.dropout = Dropout(0.3)

    def forward(self, x, edge_index, batch=None):
        """
        Forward pass through the GCN.
        
        Parameters
        ----------
        x : torch.Tensor
            Node feature matrix of shape [num_nodes, num_features]
        edge_index : torch.Tensor
            Edge indices of shape [2, num_edges]
        batch : torch.Tensor, optional
            Batch vector of shape [num_nodes] indicating which graph
            each node belongs to. If None, assumes single graph.
            
        Returns
        -------
        torch.Tensor
            Graph-level predictions of shape [num_graphs, 1]
        """
        # If batch not provided, assume single graph
        if batch is None:
            batch = torch.zeros(x.size(0), dtype=torch.long, device=x.device)

        # GCN layers
        x = self.conv1(x, edge_index)
        x = self.bn1(x)
        x = F.relu(x)

        x = self.conv2(x, edge_index)
        x = self.bn2(x)
        x = F.relu(x)

        x = self.conv3(x, edge_index)
        x = self.bn3(x)
        x = F.relu(x)

        # Global pooling
        x = global_add_pool(x, batch)

        # Fully connected layers
        x = self.dropout(F.relu(self.lin1(x)))
        x = self.lin2(x)

        return x

Usage Examples

Training a GCN

import torch
import torch.nn.functional as F
from torch_geometric.loader import DataLoader
from chemxai.models import GCN
from chemxai.data import graph_datasets

# Setup
device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')

# Load data
gd = graph_datasets()
dataset = gd.prepare_data_graph('QM9')

# Create train/val/test split
train_size = int(0.8 * len(dataset))
val_size = int(0.1 * len(dataset))
test_size = len(dataset) - train_size - val_size

train_dataset = dataset[:train_size]
val_dataset = dataset[train_size:train_size+val_size]
test_dataset = dataset[train_size+val_size:]

# Create dataloaders
train_loader = DataLoader(train_dataset, batch_size=32, shuffle=True)
val_loader = DataLoader(val_dataset, batch_size=32, shuffle=False)
test_loader = DataLoader(test_dataset, batch_size=32, shuffle=False)

# Create model
model = GCN(
    num_features=dataset[0].x.shape[1],
    hidden_dim=256
).to(device)

# Optimizer and loss
optimizer = torch.optim.Adam(model.parameters(), lr=0.001, weight_decay=1e-4)
criterion = torch.nn.MSELoss()

# Training loop
for epoch in range(100):
    model.train()
    total_loss = 0
    
    for batch in train_loader:
        batch = batch.to(device)
        optimizer.zero_grad()
        
        # Forward pass
        out = model(batch.x, batch.edge_index, batch.batch)
        
        # Compute loss (assuming regression on property at index 7)
        loss = criterion(out.squeeze(), batch.y[:, 7])
        
        # Backward pass
        loss.backward()
        optimizer.step()
        
        total_loss += loss.item() * batch.num_graphs
    
    # Validation
    model.eval()
    val_loss = 0
    with torch.no_grad():
        for batch in val_loader:
            batch = batch.to(device)
            out = model(batch.x, batch.edge_index, batch.batch)
            loss = criterion(out.squeeze(), batch.y[:, 7])
            val_loss += loss.item() * batch.num_graphs
    
    print(f'Epoch {epoch}: Train Loss = {total_loss/len(train_dataset):.4f}, '
          f'Val Loss = {val_loss/len(val_dataset):.4f}')

# Save model
torch.save(model.state_dict(), 'gcn_model.pth')

Model Comparison

Aspect MLP GCN
Input Type Fixed-size vectors (fingerprints, descriptors) Variable-size molecular graphs
Architecture Fully connected layers Graph convolutions + pooling
Advantages Simple, fast, works with any descriptor Learns from molecular structure directly
Disadvantages Requires manual feature engineering More complex, requires graph structure
Best For Quick experiments, standard descriptors Learning structural patterns, 3D data

Best Practices

MLP Best Practices

  • Layer sizes: Start with [256, 128, 64] and adjust based on data size
  • Learning rate: 0.001 is a good starting point, use learning rate scheduling for better results
  • Input normalization: Always normalize/standardize input features
  • Regularization: Add dropout or L2 regularization if overfitting occurs
  • Descriptor choice: Morgan fingerprints work well for most tasks

GCN Best Practices

  • Hidden dimensions: 256 works well for most molecular datasets
  • Number of layers: 3-5 GCN layers is typically sufficient
  • Batch size: Start with 32-64, adjust based on GPU memory
  • Pooling: global_add_pool works well for molecular properties
  • Normalization: Batch normalization helps training stability
  • Dropout: 0.3 is a good default value

Troubleshooting

MLP Issues

# Issue: Model not learning
# Solution: Check learning rate and add more layers
model = MLP(input_dim=2048, output_dim=1, 
            layers=[512, 256, 128, 64],  # More capacity
            device=device, lr=0.01)  # Higher LR

# Issue: Overfitting
# Solution: Add dropout layers manually
import torch.nn as nn
class MLPWithDropout(MLP):
    def __init__(self, *args, dropout=0.5, **kwargs):
        super().__init__(*args, **kwargs)
        # Modify layers to add dropout
        new_layers = []
        for layer in self.layers:
            new_layers.append(layer)
            if isinstance(layer, nn.ReLU):
                new_layers.append(nn.Dropout(dropout))
        self.layers = nn.Sequential(*new_layers)

GCN Issues

# Issue: Out of memory
# Solution: Reduce batch size or hidden dimension
train_loader = DataLoader(train_dataset, batch_size=16)  # Smaller batches

from chemxai.models import GCN
import torch.nn as nn

class CustomGCN(GCN):
    """Extended GCN with attention mechanism"""
    def __init__(self, num_features, hidden_dim=256, num_heads=4):
        super().__init__(num_features, hidden_dim)
        # Add custom layers here

    def forward(self, x, edge_index, batch=None):

Ensemble Methods

# Train multiple models and ensemble predictions
models = []
for i in range(5):
    model = MLP(input_dim=2048, output_dim=1, 
                layers=[256, 128, 64], device=device)
    # Train model...
    models.append(model)

# Ensemble prediction
def ensemble_predict(models, x):
    predictions = []
    for model in models:
        model.eval()
        with torch.no_grad():
            pred = model(x)
            predictions.append(pred)
    return torch.mean(torch.stack(predictions), dim=0)
